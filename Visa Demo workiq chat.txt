  'Slicer_Year':    { column: 'Year' },
  'Slicer_Country': { column: 'Country' },

  'Slicer_Product': { column: 'Product' }

applyBtn.onclick = async () => {
  const state = readHostState();   // { Year:[2014], Country:['Canada','Germany'], ... }
  const page  = (await report.getPages()).find(p => p.isActive);
  const slicers = await page.getSlicers();

  for (const s of slicers) {
    const mapping = SLICER_MAP[s.name];
    if (!mapping) continue;
    const values = state[mapping.column] || [];
    if (values.length === 0) {
      await s.setSlicerState({ filters: [] });
    } else {
      await s.setSlicerState({
        filters: [{
          $schema: "http://powerbi.com/product/schema#basic",
          target: { table: 'financials', column: mapping.column },
          operator: "In",
          values
        }]
      });
    }
  }
};
